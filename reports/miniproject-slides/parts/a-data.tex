\begin{frame}[fragile]{ETL: CSV $\rightarrow$ Parquet Data Lake}
  \begin{itemize}
    \item Đọc CSV với \textbf{schema tường minh} (\texttt{StructType}), không \texttt{inferSchema}
    \item Chuẩn hóa: \texttt{user\_id}, \texttt{movie\_id}; tách \texttt{release\_year} và mảng \texttt{genres}
    \item Ghi Parquet nén \texttt{snappy}, \textbf{phân hoạch theo năm}
    \item SparkSession: \texttt{KryoSerializer}, AQE, shuffle partitions 16 (dev) / 64 (prod)
  \end{itemize}
