\begin{frame}{Phương pháp đo}
  \begin{itemize}
    \item \textbf{Data Scaling}: huấn luyện ALS trên \SI{5}{\percent}, \SI{20}{\percent}, \SI{50}{\percent}, \SI{100}{\percent} dữ liệu
    \item \textbf{Speedup}: cùng dữ liệu, đổi \texttt{local[1]}, \texttt{local[2]}, \texttt{local[4]}
  \end{itemize}
  \[
    S_p = \frac{T_1}{T_p}, \qquad E_p = \frac{S_p}{p}, \qquad
    S_p \le \frac{1}{f + (1-f)/p} \quad \text{(Amdahl)}
  \]
  \begin{itemize}
    \item Phần tuần tự $f$: đọc Parquet, điều phối trên driver, đồng bộ giữa các vòng lặp, checkpoint ghi đĩa
    \item Kết quả ghi \texttt{benchmark\_results.csv}, biểu đồ tự sinh bằng \texttt{plot\_benchmarks.py}
  \end{itemize}
